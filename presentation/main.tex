\section{Problem CVRP}
% ===========================================================================
\begin{frame}{Czym jest CVRP?}
  \textbf{Capacitated Vehicle Routing Problem} — kanoniczny problem optymalizacji tras dostaw.
  \vfill
  \begin{columns}[T]
    \begin{column}{0.58\textwidth}
      \textbf{Dane wejściowe:}
      \begin{itemize}
        \item jeden magazyn (\emph{depot}) oraz $n$ klientów,
        \item zapotrzebowanie $q_i$ każdego klienta,
        \item flota pojazdów o wspólnej pojemności $Q$,
        \item macierz odległości (kosztów) między punktami.
      \end{itemize}
      \vspace{0.4em}
      \textbf{Cel:} zbiór tras (każda od depotu do depotu) o \textbf{minimalnej łącznej długości}, taki że:
      \begin{itemize}
        \item każdy klient odwiedzony dokładnie raz,
        \item suma zapotrzebowań na trasie $\le Q$.
      \end{itemize}
    \end{column}
    \begin{column}{0.42\textwidth}
      \centering
      \begin{tikzpicture}[scale=0.95]
        \node[rectangle,draw=accent,thick,fill=accent!20] (d) at (0,0) {Depot};
        \foreach \x/\y in {-1.4/1.2, -1.6/-0.6, 1.3/1.3, 1.7/0.1, 1.2/-1.2, -0.3/-1.5}
          \fill[ink] (\x,\y) circle (2pt);
        \draw[good,thick] (d) -- (-1.4,1.2) -- (-1.6,-0.6) -- (d);
        \draw[bad,thick] (d) -- (1.3,1.3) -- (1.7,0.1) -- (d);
        \draw[accent2,thick] (d) -- (1.2,-1.2) -- (-0.3,-1.5) -- (d);
      \end{tikzpicture}
      \par\vspace{0.3em}{\scriptsize Trzy trasy obsługujące sześciu klientów.}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{Dlaczego to trudny problem?}
  \begin{itemize}
    \item CVRP jest \textbf{NP-trudny} — uogólnia problem komiwojażera (TSP).
    \item Liczba możliwych podziałów i kolejności rośnie wykładniczo z liczbą klientów.
    \item Dla instancji z tysiącami klientów metody dokładne są bezużyteczne — stosuje się \textbf{metaheurystyki}.
    \item Znaczenie praktyczne: logistyka, kurierzy, dystrybucja, odbiór odpadów — poprawa o ułamek procenta na dużej flocie to realne oszczędności paliwa i czasu.
  \end{itemize}
  \vfill
  \begin{block}{Punkt odniesienia}
    \textbf{HGS} (Hybrid Genetic Search, Vidal i in.) to jeden z najlepszych znanych algorytmów dla CVRP — bardzo dobre rozwiązania w krótkim czasie. Naszą pracę budujemy jako jego rozszerzenie (fork).
  \end{block}
\end{frame}

\section{Algorytm HGS}
% ===========================================================================
\begin{frame}{HGS w skrócie: ewolucja + intensywna lokalna poprawa}
  HGS łączy \textbf{algorytm genetyczny} (eksploracja przestrzeni) z \textbf{przeszukiwaniem lokalnym} (eksploatacja okolicy rozwiązania). Pięć współpracujących komponentów:
  \vfill
  \begin{columns}[T]
    \begin{column}{0.5\textwidth}
      \begin{enumerate}
        \item \textbf{Reprezentacja + Split} — rozwiązanie jako permutacja klientów, dzielona na trasy.
        \item \textbf{Krzyżowanie (OX)} — z dwóch rodziców powstaje potomek.
        \item \textbf{Przeszukiwanie lokalne} — RI (relokacje, zamiany, 2-opt) oraz SWAP*.
      \end{enumerate}
    \end{column}
    \begin{column}{0.5\textwidth}
      \begin{enumerate}
        \setcounter{enumi}{3}
        \item \textbf{Populacja z różnorodnością} — podpopulacje dopuszczalna/niedopuszczalna, selekcja wg \emph{biased fitness}.
        \item \textbf{Adaptacyjne kary} — sterują dopuszczalnością przez całą optymalizację.
      \end{enumerate}
    \end{column}
  \end{columns}
  \vfill
  {\footnotesize Kluczowa cecha: pętla jest \textbf{sekwencyjna i adaptacyjna} — każda iteracja zależy od stanu poprzednich. To później okaże się istotne dla zrównoleglenia.}
\end{frame}

\begin{frame}{Reprezentacja rozwiązania i algorytm Split}
  \begin{columns}[T]
    \begin{column}{0.56\textwidth}
      \textbf{Gigantyczna trasa (\emph{giant tour}).} Rozwiązanie to \textbf{permutacja klientów} $1\ldots n$ — sama kolejność odwiedzin, \emph{bez} przypisania do pojazdów. Krzyżowanie działa właśnie na tej permutacji.
      \vspace{0.5em}

      \textbf{Split.} Optymalnie tnie permutację na trasy metodą \textbf{programowania dynamicznego} (Bellman po grafie acyklicznym):
      \begin{itemize}
        \item wariant \textbf{$O(n)$} bez ograniczeń czasowych (Vidal 2016),
        \item wariant z limitem floty (\texttt{splitLF}) i z ograniczeniem czasu trasy.
      \end{itemize}
      Wynik: konkretne trasy + koszt z naruszeniami.
    \end{column}
    \begin{column}{0.44\textwidth}
      \centering
      \begin{tikzpicture}[node distance=3mm]
        \node[box,fill=accent!12] (gt) {giant tour: \;3\,1\,5\,2\,4\,6};
        \node[fb, below=7mm of gt] (sp) {Split (DP)};
        \node[box, below left=3mm and 2mm of sp] (r1) {trasa A: 3\,1};
        \node[box, below=10mm of sp] (r2) {trasa B: 5\,2};
        \node[box, below right=3mm and 2mm of sp] (r3) {trasa C: 4\,6};
        \draw[fl] (gt)--(sp);
        \draw[fl] (sp)--(r1); \draw[fl] (sp)--(r2); \draw[fl] (sp)--(r3);
      \end{tikzpicture}
      \par\vspace{0.2em}{\scriptsize Split rozdziela koszt krzyżowania od struktury tras.}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{Przeszukiwanie lokalne — sąsiedztwo RI}
  Po Splicie potomek jest intensywnie ulepszany. Rozwiązanie trzymane jest jako \textbf{lista dwukierunkowa} węzłów (\texttt{Node}) i tras (\texttt{Route}) — szybkie wstawianie i usuwanie klientów.
  \vfill
  \begin{columns}[T]
    \begin{column}{0.54\textwidth}
      \textbf{Ruchy RI} (klasyczny zestaw Prinsa, \texttt{move1}--\texttt{move9}):
      \begin{itemize}
        \item relokacja 1 lub 2 klientów,
        \item zamiana klientów (swap),
        \item 2-opt (wewnątrz trasy) i 2-opt* (między trasami).
      \end{itemize}
    \end{column}
    \begin{column}{0.46\textwidth}
      \textbf{Dwa mechanizmy przyspieszające:}
      \begin{itemize}
        \item \textbf{Sąsiedztwo granularne} — ruch rozważany tylko między klientem a jego \texttt{nbGranular} najbliższymi (\texttt{correlatedVertices}).
        \item \textbf{,,Don't-look''} — znaczniki \texttt{whenLastTested} + licznik \texttt{nbMoves} pomijają ponowną ocenę niezmienionych węzłów.
      \end{itemize}
    \end{column}
  \end{columns}
  \vfill
  {\footnotesize Local search jest \textbf{najdroższym krokiem} całej tury — i głównym celem naszego zrównoleglania.}
\end{frame}

\begin{frame}{Sąsiedztwo SWAP* — to akcelerujemy na GPU}
  \begin{columns}[T]
    \begin{column}{0.6\textwidth}
      Dla \textbf{pary tras} $(U,V)$ SWAP* rozważa wymianę klienta z $U$ na klienta z $V$ — ale \textbf{każdy z nich wstawiany jest w najlepsze miejsce} drugiej trasy (nie w pozycję wymienianego).
      \vspace{0.5em}
      \begin{itemize}
        \item \texttt{ThreeBestInsert} — trzy najtańsze pozycje wstawienia każdego klienta (z pominięciem sąsiedztwa usuwanego).
        \item \textbf{Przycinanie}: rozważane tylko pary tras o \textbf{nakrywających się sektorach kątowych} (\texttt{CircleSector}).
        \item Koszt ruchu $=$ usunięcia $+$ najtańsze wstawienia $+$ zmiana kar.
      \end{itemize}
    \end{column}
    \begin{column}{0.4\textwidth}
      \centering
      \begin{tikzpicture}[scale=0.95]
        \node[font=\scriptsize] at (-1.1,1.5) {trasa $U$};
        \node[font=\scriptsize] at ( 1.1,1.5) {trasa $V$};
        \foreach \y in {0.8,0.2,-0.4,-1.0} \fill[good] (-1.1,\y) circle (2.2pt);
        \foreach \y in {0.8,0.2,-0.4,-1.0} \fill[accent2] (1.1,\y) circle (2.2pt);
        \draw[fl,draw=bad] (-1.1,0.2) .. controls (0,0.7) .. (1.1,-0.4);

        \node[font=\tiny,bad] at (0,1.05) {najlepsze wstawienie};
      \end{tikzpicture}
      \par\vspace{0.2em}{\scriptsize Liczba ocen rośnie z \textbf{kwadratem} długości trasy — stąd masa pracy dla GPU.}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{Populacja, różnorodność i selekcja}
  \begin{columns}[T]
    \begin{column}{0.55\textwidth}
      \textbf{Dwie podpopulacje}, każda sortowana wg kosztu z karami:
      \begin{itemize}
        \item \textbf{dopuszczalna} (spełnia pojemność/czas),
        \item \textbf{niedopuszczalna} (z naruszeniami).
      \end{itemize}
      \vspace{0.4em}
      \textbf{Biased fitness} łączy dwa rankingi:
      \[
        f_{\text{bias}} = \underbrace{r_{\text{koszt}}}_{\text{jakość}}
          + \Bigl(1-\tfrac{n_{\text{elite}}}{|P|}\Bigr)\underbrace{r_{\text{div}}}_{\text{różnorodność}}
      \]
      Różnorodność = \emph{broken-pairs distance} do \texttt{nbClose} najbliższych rozwiązań.
    \end{column}
    \begin{column}{0.45\textwidth}
      \textbf{Selekcja ocalałych.} Gdy podpopulacja przekroczy \texttt{mu+lambda}, przycinana jest \textbf{do \texttt{mu}}: usuwany najgorszy wg biased fitness, z \textbf{preferencją usuwania klonów}.
      \vspace{0.5em}
      \begin{block}{Po co różnorodność?}
        \footnotesize Sama jakość prowadzi do przedwczesnej zbieżności (klony). Premiowanie odmiennych rozwiązań utrzymuje eksplorację.
      \end{block}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{Adaptacyjne kary, krzyżowanie i restart}
  \begin{itemize}
    \item \textbf{Kary} \texttt{penaltyCapacity} i \texttt{penaltyDuration} są adaptowane co \texttt{nbIterPenaltyManagement} iteracji w stronę docelowego odsetka rozwiązań dopuszczalnych (\texttt{targetFeasible} $\approx 0{,}2$): za mało dopuszczalnych $\Rightarrow$ kary rosną, za dużo $\Rightarrow$ maleją. Pozwala to kontrolowanie naruszeń \emph{przejściowo}.
    \item \textbf{Krzyżowanie OX} (ordered crossover) na gigantycznych trasach: kopiuje wycinek pierwszego rodzica, resztę uzupełnia w kolejności drugiego.
    \item \textbf{Zakończenie / restart}: brak poprawy przez \texttt{nbIter} iteracji kończy bieg; przy limicie czasu populacja jest restartowana i optymalizacja trwa do wyczerpania czasu.
  \end{itemize}
\end{frame}

\section{Wprowadzone modyfikacje}
% ===========================================================================
\begin{frame}{Trzy osie zrównoleglenia}
  Do HGS dodaliśmy trzy niezależne (i łączalne) mechanizmy:
  \vfill
  \begin{columns}[T]
    \begin{column}{0.33\textwidth}
      \begin{block}{1. GPU SWAP* (CUDA)}
        \footnotesize Ewaluacja par tras dla SWAP* przeniesiona na GPU; jeden blok na parę, strumienie CUDA.
      \end{block}
    \end{column}
    \begin{column}{0.33\textwidth}
      \begin{block}{2. Master--slave (OpenMP)}
        \footnotesize Równoległe wytwarzanie potomstwa: 16 potomków na rundę na 8 wątkach.
      \end{block}
    \end{column}
    \begin{column}{0.33\textwidth}
      \begin{block}{3. Model wysp (MPI)}
        \footnotesize 8 niezależnych populacji, topologia gwiazdy, polityki migracji i selekcji migrantów.
      \end{block}
    \end{column}
  \end{columns}
  \vfill
  Dodatkowo: warstwa \textbf{telemetrii} i \textbf{monitor sprzętu} (CPU/GPU, NVML) do rzetelnej oceny.
  \vfill
  {\footnotesize Sensownie przetestowaliśmy \textbf{6 wariantów}: \texttt{baseline}, \texttt{gpu}, \texttt{offspring16t8}, \texttt{gpu\_offspring16t8}, \texttt{island8\_star}, \texttt{island8\_star\_gpu\_off16t8}.}
\end{frame}

%  OŚ 1 — GPU SWAP*

\begin{frame}{Oś 1: SWAP* na GPU — pomysł}
  \textbf{Co przyspieszamy?} W każdym przebiegu local search SWAP* ocenia \textbf{ogromną liczbę par klientów} z różnych tras. To wiele niezależnych, identycznych obliczeń — \emph{idealny wzorzec dla GPU}, które liczy tysiące takich par jednocześnie.
  \vfill
  \centering
  \begin{tikzpicture}[node distance=14mm]
    \node[box, fill=accent!12] (host) {\textbf{Host (CPU)}\\\scriptsize pętla LS,\\bieżące trasy};
    \node[draw=accent2!70!black, thick, rounded corners, fill=accent2!8, right=34mm of host,
          minimum width=30mm, minimum height=22mm, label={[font=\scriptsize]above:\textbf{GPU} — siatka bloków/wątków}] (gpu) {};
    \foreach \r in {0,1,2}\foreach \c in {0,1,2,3,4}
      \node[cell] at ($(gpu.center)+(-1.0+\c*0.5,0.55-\r*0.55)$) {};
    \draw[fl] (host.north east) ++(0,1mm) .. controls +(12mm,7mm) and +(-12mm,7mm) ..
          node[above,font=\tiny]{kopiuj \textbf{trasy i pary}} (gpu.north west);
    \draw[fl, draw=good!70!black] (gpu.south west) .. controls +(-12mm,-7mm) and +(12mm,-7mm) ..
          node[below,font=\tiny]{zwróć najlepsze ruchy} (host.south east);
  \end{tikzpicture}
  \vfill
  {\footnotesize Jeden \textbf{blok GPU} przetwarza jedną parę tras; CPU dostaje gotowy zestaw najlepszych zamian do zastosowania.}
\end{frame}

\begin{frame}[fragile]{Oś 1: SWAP* na GPU — implementacja}
  \textbf{Co zbudowaliśmy (CUDA):}
  {\small
  \begin{itemize}
    \item \textbf{Kernel} — jeden blok CUDA na parę tras (256 wątków); wątki oceniają kandydatów SWAP*/relokacji, a \textbf{redukcja warp + shared memory} wyłania najlepszy ruch pary.
    \item \textbf{Spłaszczony układ danych} (\texttt{GpuDataLayout}) — trasy, klienci, koszty usunięć i pary w jednym buforze (koalescencja dostępu).
    \item \textbf{Współdzielony kontekst} (\texttt{GpuProblemData}) — macierz odległości wgrana \textbf{raz na proces} (read-only), referencja we wszystkich instancjach.
    \item \textbf{Strumień CUDA na instancję} — asynchroniczne transfery i synchronizacja \emph{własnego} strumienia $\Rightarrow$ równoległe instancje LS \textbf{nakładają się} na GPU. Pomiar zużycia przez \textbf{NVML}.
  \end{itemize}}
  \vspace{0.1em}

\end{frame}

\begin{frame}{Oś 1: SWAP* na GPU — kompromis}
  \begin{itemize}
    \item Karta jest wydajna tylko przy \textbf{dużej ilości równoległej pracy na jedno wywołanie}. Liczba par rośnie z kwadratem długości trasy, więc \textbf{krótkie trasy} (dużo pojazdów) dają mało pracy i dominuje \textbf{narzut uruchomienia kernela i transferu}.
    \item \textbf{Realny zysk} pojawia się dopiero przy długich trasach lub wielu równoległych ocenach naraz — co pokażą wyniki.
  \end{itemize}
\end{frame}

\begin{frame}{Oś 2: model master--slave — pomysł}
  \textbf{Co przyspieszamy?} Najdroższy krok tury to local search potomka. Skoro mamy wiele rdzeni CPU — \textbf{produkujemy wielu potomków naraz}, każdy ulepszany na osobnym wątku.

  \vfill
  {\footnotesize Nasz wariant: \textbf{16 potomków na rundę, 8 wątków}. Master zbiera potomków i seryjnie wstawia do populacji.}

\begin{frame}[fragile]{Oś 2: model master--slave — implementacja}
  \textbf{Co zbudowaliśmy (OpenMP):}
  \begin{itemize}
    \item Wspólny interfejs \texttt{OffspringMaker} z dwiema implementacjami: \texttt{Standard} (sekwencyjna) i \texttt{Parallel}.
    \item \texttt{ParallelOffspringMaker} — pula wątków; każdy wątek wykonuje pełny cykl: \textbf{selekcja turniejowa $\to$ OX $\to$ Split $\to$ local search}, na \textbf{własnym} \texttt{LocalSearch} i \texttt{Split} (brak współdzielenia stanu, brak wyścigów).
  \end{itemize}
  \vspace{0.3em}
\end{frame}

\begin{frame}{Wnioski}
  \begin{block}{Główny wniosek}
    Zrównoleglenie \textbf{nie poprawiło jakości} w istotny sposób, a najbardziej złożony wariant często ją pogarszał. Wąskim gardłem nie jest przepustowość obliczeń, lecz \textbf{sam algorytm i struktura zależności danych}.
  \end{block}
  \vfill
  \begin{itemize}
    \item HGS jest zbudowany wokół \textbf{sekwencyjnej, adaptacyjnej pętli} (kary, różnorodność, selekcja), która nie rozbija się na niezależne kawałki bez zmiany dynamiki przeszukiwania.
    \item GPU pomaga \textbf{tylko} tam, gdzie trasy są długie (dużo równoległej pracy); przy krótkich trasach narzut przewyższa zysk.
    \item Lepszym kierunkiem byłyby algorytmy iteracyjnej poprawy \textbf{zaprojektowane z myślą o równoległości}, zamiast doklejania równoległości do z gruntu sekwencyjnego schematu.
  \end{itemize}
\end{frame}
